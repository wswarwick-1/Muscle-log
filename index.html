if (rir <= 1 && reps < targetReps) return lastWeight - 5; return lastWeight; } ///////////////////////////// // 🗂 GLOBAL STORE ///////////////////////////// const useStore = create((set) => ({ logs: [], fatigue: 0, addSet: (setData) => set((state) => ({ logs: [setData, ...state.logs] })), updateFatigue: (rir) => set((state) => ({ fatigue: state.fatigue + (5 - rir) })), reset: () => set({ logs: [], fatigue: 0 }) })); ///////////////////////////// // 🎨 STYLES ///////////////////////////// const colors = { bg: "#121212", card: "#1e1e1e", accent: "#FFD60A", text: "#ffffff", sub: "#aaaaaa" }; ///////////////////////////// // 🏋️ MAIN APP ///////////////////////////// export default function App() { const { logs, fatigue, addSet, updateFatigue, reset } = useStore(); const entry = { id: Date.now().toString(), exercise: "Bench Press", weight: Number(weight), reps: Number(reps), rir: Number(rir), nextWeight }; addSet(entry); updateFatigue(Number(rir)); setWeight(""); setReps(""); setRir(""); }; return ( {/* HEADER */} Hypertrophy Tracker Fatigue Score: {fatigue} {/* INPUT CARD */} Bench Press Log Set {/* LOG LIST */} item.id} renderItem={({ item }) => ( {item.exercise} {item.weight}kg × {item.reps} reps (RIR {item.rir}) Next: {item.nextWeight}kg )} /> {/* RESET BUTTON */} Reset Workout ); } ///////////////////////////// // ✨ REUSABLE STYLES ///////////////////////////// const inputStyle = { backgroundColor: "#2a2a2a", color: "#fff", padding: 10, borderRadius: 8, marginBottom: 10 }; const buttonStyle = { backgroundColor: colors.accent, padding: 12, borderRadius: 10, alignItems: "center" };
